\beat{1}{The LP relaxation --- and the question it leaves}
\begin{board}
\textbf{LP relaxation}: delete the integrality line --- optimize over $P$ instead of $X = P \cap \mathbb{Z}^n$
\[
  (\mathrm{IP}) \ \ \max_{x} \{\, c^{\top}x : Ax \le b, \ x \in \mathbb{Z}^n_{+} \,\}
  \qquad\longrightarrow\qquad
  (\mathrm{LP}) \ \ \max_{x} \{\, c^{\top}x : Ax \le b, \ x \ge 0 \,\}
\]
larger feasible set $\Rightarrow$ always optimistic: \qquad
$z_{\mathrm{LP}} \ge z_{\mathrm{IP}}$ \ (max), \qquad $z_{\mathrm{LP}} \le z_{\mathrm{IP}}$ \ (min)
\vspace{3pt}

week 2: \ $22.5 \ge 19$ on the toy, rounding worth only $18$ --- a \textbf{bound}, not an answer. bounds solve every IP (week 7)
\vspace{3pt}

\textbf{today: when is the gap zero?}
\vspace{2pt}

\quad gap $= 0$ $\Rightarrow$ write $x \in \mathbb{Z}$, solve the relaxation, get an integer plan anyway --- \emph{the LP \textbf{is} the IP}
\vspace{1pt}

\quad not luck, not rounding: a whole family (assignment, transportation, networks), readable off $A$
\end{board}
\key{one line, $\in \mathbb{Z}$, is what makes an IP hard --- so a model where deleting it costs nothing is worth recognizing on sight.}

\beat{2}{Assignment --- a model born canonical}
\demo{ip-games / assignment}{play FIRST: four cadets, four branches, one each --- the board below is the same model at $n = 3$. \ \url{https://hankpark0706.github.io/ip-games/assignment-matching.html}}
\begin{board}
3 pilots, 3 aircraft, one each --- officers to billets, units to sectors: the \textbf{assignment problem} {\footnotesize(할당 문제)}
\vspace{2pt}

$c_{ij}$ = value of pairing $(i,j)$; \ the decision is a \emph{table}: \quad
$X \in \{0,1\}^{3\times 3}$, \quad $x_{ij} = 1$ iff pilot $i$ flies aircraft $j$
\vspace{2pt}

rows sum to 1 (each pilot once), columns sum to 1 (each aircraft once) --- week 2's BP:
\[
\begin{aligned}
  \max_X\quad & \langle C, X \rangle \\
  \text{s.t.}\quad & X \mathbf{1} = \mathbf{1}, \quad X^{\top}\mathbf{1} = \mathbf{1} \\
                   & x_{ij} \in \{0,1\}
\end{aligned}
\]
\emph{read the symbols:} \ $\langle C, X \rangle = \sum_{i,j} c_{ij} x_{ij}$, \quad $\mathbf{1}$ = all-ones vector
\vspace{2pt}

\textbf{flatten}: $x = \operatorname{vec}(X) \in \mathbb{R}^9$ \ $\Rightarrow$ \ $A$ is $6 \times 9$, all 0/1, \ $b = \mathbf{1}$
\vspace{1pt}

\quad \textbullet\ each column: exactly \textbf{two ones} --- one pilot row, one aircraft row \ (an \emph{incidence matrix})
\vspace{1pt}

\quad \textbullet\ $\operatorname{rank} A = 5$: any one row is implied by the rest
\vspace{3pt}

\textbf{relax} (section 1): \ $x_{ij} \in \{0,1\} \to x_{ij} \ge 0$ (row sums already cap at 1)
\ $\Rightarrow$ \ the LP returns a \textbf{matching}, every cost table, every time
\end{board}

\beat{3}{Transportation --- assignment with quantities}
\begin{board}
supply $s_i$ at source $i$, demand $d_j$ at sink $j$ (assignment: all $= 1$) \ $\Rightarrow$ \ matching becomes \textbf{flow}
\vspace{2pt}

$x_{ij}$ = amount shipped $i \to j$ --- crates, pallets, sorties: whole units, so integrality is declared
\[
\begin{aligned}
  \min_{x}\quad & \textstyle\sum_{i=1}^{n} \sum_{j=1}^{k} c_{ij}\, x_{ij} \\
  \text{s.t.}\quad
    & \textstyle\sum_{j=1}^{k} x_{ij} \le s_i && (i \in [n]) && \text{ship no more than you hold \ \footnotesize(공급 한도)} \\
    & \textstyle\sum_{i=1}^{n} x_{ij} \ge d_j && (j \in [k]) && \text{every sink gets what it needs \ \footnotesize(수요 충족)} \\
    & x_{ij} \in \mathbb{Z}_{+}
\end{aligned}
\]
\textbf{balanced}: \ $\sum_i s_i = \sum_j d_j$ \qquad
with $c \ge 0$ nobody ships spare, so the $\ge d_j$ rows may be written $=$
\vspace{3pt}

\textbf{relax}: \ $x_{ij} \in \mathbb{Z}_{+} \to x_{ij} \ge 0$ \ $\Rightarrow$ \ integer $s, d$ give whole shipments back.
\ gap zero again --- and not a binary in sight
\vspace{2pt}

\quad assignment is the special case $k = n$, \ $s_i = d_j = 1$: row sums cap $x_{ij}$ at 1, and section 4 puts the
optimum on a \emph{corner} --- a permutation matrix
\end{board}
\takeaway{transportation = assignment with quantities: supplies left, demands right, flows between. an IP you may hand to an LP solver whenever the data is whole.}

% ---------------------------------------------------------------- 4
\beat{4}{Total unimodularity --- why the LP told the truth}
\begin{board}
gap on week 1's toy: $22.5 - 19$. \ gap on assignment, transportation: $0$. \ same solver $\Rightarrow$ the difference is in the \emph{matrix}
\vspace{3pt}

\textbf{Def.} \ $A$ is \textbf{totally unimodular (TU)} {\footnotesize(완전 유니모듈러 행렬)}: every square submatrix has
determinant $0$, $+1$, or~$-1$
\vspace{1pt}

\quad (any $k$ rows $\times$ any $k$ columns, every $k$) \quad --- a property of $A$ alone: no $b$, no $c$
\vspace{3pt}

\textbf{$k = 1$ is free:} \ a $1 \times 1$ submatrix is one entry \ $\Rightarrow$ \ \textbf{every entry of a TU matrix is $0, \pm 1$}
\vspace{1pt}

\quad week 2's rucksack row $(5,7,8,8,8,2)$ has an 8 $\Rightarrow$ not~TU \quad (the numbers, not the shape: $\sum_i x_i \le p$ is TU)
\vspace{3pt}

\textbf{Prop.} (Hoffman--Kruskal) \ $A$ TU \textbf{and} $b$ integer \ $\Rightarrow$ \ every corner of
$P = \{x : Ax \le b, \ x \ge 0\}$ is integer (same for $Ax = b$)
\vspace{1pt}

\quad $\Rightarrow$ \ the LP, if it has an optimum, has an \emph{integer} one: \ \textbf{the relaxation solves the IP}
\vspace{3pt}

\emph{mechanism}: in $Ax = b$, $x \ge 0$ a corner picks \emph{basic columns} $B$ (the rest are 0), and $A_B x_B = b$.
Cramer:
\[
  (x_B)_j \;=\; \frac{\det(A_B \text{ with column } j \text{ replaced by } b)}{\det A_B}
\]
integer on top; \ $\det A_B = \pm 1$ \ $\Rightarrow$ \ no fraction can appear. \ ($\det = 0$: not a basis, never a corner)
\end{board}
\key{TU is the promise that \emph{no} corner can be fractional, whatever integer $b$ you throw at it.}

\begin{center}
\begin{tabular}{@{}l|l@{}}

\end{tabular}
\end{center}

\beat{6}{Minimum cost network flow (MCNF) --- the family worth recognizing}
\demo{ip-games / min-cost-flow}{push flow through a network by hand, then let the solver do it. \ \url{https://hankpark0706.github.io/ip-games/min-cost-network-flow.html}}
\begin{board}
directed graph $G = (V,E)$, \ $|V| = m$ nodes {\footnotesize(노드)}, arcs {\footnotesize(아크)} $(i,j) \in E$
\vspace{2pt}

\textbf{node $i$}: \ $b_i$ = net supply = outflow $-$ inflow \qquad\quad
\textbf{arc $(i,j)$}: \ cost $c_{ij}$, \ bounds $l_{ij} \le x_{ij} \le u_{ij}$ \quad
\textbf{decision}: \ $x_{ij}$ = flow (whole units $\Rightarrow$ integer)
\begin{center}
\begin{tabular}{@{}l|l|l@{}}
 $b_i > 0$ & node $i$ supplies & \textbf{source} {\footnotesize(공급지)} \\[2pt]
 $b_i < 0$ & node $i$ demands & \textbf{destination} {\footnotesize(수요지)} \\[2pt]
 $b_i = 0$ & flow only passes through & \textbf{transhipment}, a hub {\footnotesize(환적지)} \\
\end{tabular}
\end{center}
\[
\begin{aligned}
  \min_{x}\quad & \textstyle\sum_{(i,j) \in E} c_{ij}\, x_{ij} \\
  \text{s.t.}\quad
    & \underbrace{\textstyle\sum_{j \,:\, (i,j) \in E} x_{ij} \;-\; \sum_{j \,:\, (j,i) \in E} x_{ji} \;=\; b_i}
        _{\text{one row per node --- \textbf{flow conservation} (흐름 보존)}}
      \quad (\forall i \in V) \\[4pt]
    & l_{ij} \le x_{ij} \le u_{ij}, \quad x_{ij} \in \mathbb{Z}_{+}
\end{aligned}
\]
rows sum to zero $\Rightarrow$ $\sum_{i} b_i = 0$ required --- else add a zero-cost \textbf{dummy node}
\vspace{2pt}

\textbf{node--arc incidence matrix}: one row per node, one column per arc, $+1$ where the arc leaves, $-1$ where it arrives
\end{board}

\begin{board}
checklist: \ (i) entries $0,\pm 1$ \checkmark \quad (ii) two per column \checkmark \quad
(iii) $M_1 = V$, $M_2 = \emptyset$ \checkmark \ --- \emph{every} directed graph passes
\vspace{1pt}

\quad (undirected: two $+1$s need a bipartite split --- the \textbf{matching LP} is integral only on bipartite graphs)
\vspace{1pt}

\quad $\operatorname{rank} A = |V| - 1 = 3$: one conservation row is always redundant
\vspace{3pt}

\textbf{Prop.} \ integer $\{b_i\}$ and integer bounds $\{l_{ij}\}, \{u_{ij}\}$ \ $\Rightarrow$ \ every extreme point is
integral: delete $x \in \mathbb{Z}_{+}$, solve the LP, \textbf{no branching, ever}
\vspace{3pt}

\textbf{one model, four classics:}
\begin{center}
\begin{tabular}{@{}l|l@{}}
 problem & is MCNF with \\ \hline
 transportation & sources $b_i = s_i > 0$, \ destinations $b_j = -d_j$ \\[2pt]

\end{board}
\key{recognize an IP as a network and it is solved: one row per node, one column per arc, $\pm 1$ entries --- TU by construction, so the relaxation is free.}

\beat{7}{Facility location --- paying to open}
\begin{board}
transportation assumed the depots existed --- \emph{which depots should exist?} {\footnotesize(시설 입지 문제)}
\vspace{2pt}

\emph{watch the letters:} back to week 2's convention --- $x$ = integer (open or not), $y$ = continuous.
relief tons are divisible, so the flow is no longer the integer decision
\vspace{2pt}

$x_i = 1$ iff depot $i$ opens, \ $f_i$ = fixed opening cost, \ $y_{ij}$ = tons shipped
\[
\begin{aligned}
  \min_{x,\,y}\quad & \textstyle\sum_{i} f_i x_i \;+\; \sum_{i} \sum_{j} c_{ij} y_{ij}
    && \text{fixed $+$ shipping} \\
  \text{s.t.}\quad
    & \textstyle\sum_{i} y_{ij} = d_j && \text{(demand)} \\
    & y_{ij} \le M x_i && \textbf{(linking --- week 2)} \\
    & y_{ij} \ge 0, \quad x_i \in \{0,1\}
\end{aligned}
\]
$x_i = 0 \Rightarrow y_{ij} = 0$; \ $x_i = 1$ releases it --- \emph{un}capacitated
\vspace{1pt}

\quad \emph{how big is $M$?} \ $y_{ij} \le d_j$ always $\Rightarrow$ $M = d_j$, smallest valid. \
$M = \sum_j d_j$: valid, but on section 6's network the LP bound drops $52 \to 47$
\vspace{3pt}

\textbf{1. fix $x$} $\Rightarrow$ transportation falls out: \emph{all the hardness is in the binaries}
\vspace{2pt}

\textbf{2. big-$M$ costs the guarantee:} $y_{ij} \le M x_i$ belongs to no incidence matrix $\Rightarrow$ MCNF's proposition is
gone. \ (not TU $\ne$ the LP lies: only \emph{some} $b$ must go fractional. section 8 checks)
\end{board}
\key{every guarantee in this note is a property of $A$ --- so every one is a single modeling decision away from being lost.}

% ---------------------------------------------------------------- 8
\beat{8}{The LAP --- when $b$ is a decision}
\begin{board}
MCNF takes $b_i$ as data; the \textbf{location--allocation problem} {\footnotesize(입지-배분 문제)} chooses it
\vspace{2pt}

$r_i \ge 0$ = reserve held at $i$ (cost $h_i$/ton), \quad $q_i \ge 0$ = unmet demand (penalty $u_i$), \quad
$s_i \ge 0$ = surplus (penalty $o_i$; not section 3's supply)
\[
  b_i \;=\; \underbrace{r_i}_{\text{decision}} - \; d_i + q_i - s_i
  \qquad\Longrightarrow\qquad
  \underbrace{\textstyle\sum_{j} y_{ji} - \sum_{j} y_{ij}}_{\text{inflow} \,-\, \text{outflow}}
  \;+\; r_i + q_i - s_i \;=\; d_i \quad (\forall i \in V)
\]
\[
\begin{aligned}
  \min_{x,\,r,\,y,\,q,\,s}\quad
    & \textstyle\sum_{i} f_i x_i + \sum_{i} h_i r_i + \sum_{(i,j) \in E} c_{ij} y_{ij}
      + \sum_{i} u_i q_i + \sum_{i} o_i s_i \\
  \text{s.t.}\quad
    & \textstyle\sum_{j} y_{ji} - \sum_{j} y_{ij} + r_i + q_i - s_i = d_i && (\forall i) && \text{[conservation --- MCNF]} \\
    & \textstyle\sum_{i \in S} r_i \le B && && \text{[budget --- a knapsack row]} \\
    & r_i \le B\, x_i && (\forall i \in S) && \text{[linking --- week 2]} \\
    & \textstyle\sum_{i \in S} x_i \le p && && \text{[at most $p$ depots --- week 2]} \\
    & y, r, s \ge 0, \quad 0 \le q_i \le d_i, \quad x_i \in \{0,1\}
\end{aligned}
\]
five costs: open ($f_i$), hold ($h_i$), ship ($c_{ij}$), fail ($u_i$), waste ($o_i$). \ every constraint is old
\vspace{1pt}

\quad $S \subseteq V$ = \textbf{candidate sites}; elsewhere $x_i = r_i = 0$ --- no reserve inside the disaster area
\vspace{1pt}

\quad $M = B$ passes, since the budget row already caps each $r_i$ at $B$ --- valid, though week 2 would ask for the
\emph{smallest} valid $M$
\end{board}
\begin{board}
section 6's network, priced --- $S = \{1,2\}$, $B = 15$, $p = 1$, $h_i = 1$, $u_i = 100$, $o_i = 1$:
\[
\begin{aligned}
  \min\quad & 10 x_1 + 20 x_2 + r_1 + r_2 + 2 y_{13} + y_{23} + 3 y_{34} + 4 y_{14}
    + 100 (q_3 + q_4) + \textstyle\sum_i s_i \\
  \text{s.t.}\quad
    & -y_{13} - y_{14} + r_1 + q_1 - s_1 = 0 && \text{[node 1]} \\
    & -y_{23} + r_2 + q_2 - s_2 = 0 && \text{[node 2]} \\
    & y_{13} + y_{23} - y_{34} + q_3 - s_3 = 5 && \text{[node 3]} \\
    & y_{34} + y_{14} + q_4 - s_4 = 8 && \text{[node 4]} \\
    & r_1 + r_2 \le 15, \qquad r_1 \le 15 x_1, \qquad r_2 \le 15 x_2, \qquad x_1 + x_2 \le 1 \\
    & y, r, s \ge 0, \quad q_1 = q_2 = 0, \quad 0 \le q_3 \le 5, \quad 0 \le q_4 \le 8, \quad x_1, x_2 \in \{0,1\}
\end{aligned}
\]
sixteen variables, two binary ($q_1 = q_2 = 0$ since $d_1 = d_2 = 0$, so fourteen move). \ at $p = 1$, two plans:

site 2 ships cheaper and still loses: $-5$ on the road, $+10$ at the door

\vspace{2pt}

\end{board}
